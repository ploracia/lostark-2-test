## 6. 캐릭터별 이관 확인 페이지 설계용 분석 (Open API 매핑 & 수동 입력)

### 6.1 권장 API 호출 전략
- **초기 캐릭터 1명 조회**:
  1. `GET /armories/characters/{characterName}` (통합 1회 호출, 또는 `?filters=profiles+equipment+engravings+gems+arkpassive+combat-skills+cards`) — 전체 스펙 확보.
  2. `GET /characters/{characterName}/siblings` (선택·병렬 1회) — 원정대 내 캐릭터 전환, 6캐릭/7번째 캐릭 골드 기준 안내, 원정대 보석 공유 시뮬레이션용.
- **주의**: 한글 캐릭터명 `encodeURIComponent`, 분당 100회 Rate Limit 대응 캐싱, 없는 캐릭터(`200 + null` 또는 `404`) 방어 필수.

### 6.2 입력값 ↔ 로스트아크 Open API 필드 매핑표

#### A. Open API로 자동 조회 가능한 값 (장착 기준 자동 계산 가능)

// 3. 16부위 이관 결과 및 신규 평균 아이템 레벨 계산

function calculate16Slots(oldSlots: {
  helm: { ilvl: number; honing: number };
  shoulder: { ilvl: number; honing: number }; // API: "어깨"
  chest: { ilvl: number; honing: number };
  pants: { ilvl: number; honing: number };
  gloves: { ilvl: number; honing: number };
  normalWeapon?: { ilvl: number; honing: number };
  estherWeapon?: { ilvl: number; evolution: number; ella: number };
  oldVambraceHoning?: number; // 기존 완갑 재련 단계 (API 노출 여부 미확인)
}) {
  // 1) 방어구 5부위 -> 좌 5부위 + 우 5부위 (총 10부위)
  const armorPairs = [
    ["투구", "목걸이", oldSlots.helm],
    ["견갑", "귀걸이1", oldSlots.shoulder],
    ["상의", "귀걸이2", oldSlots.chest],
    ["하의", "반지1", oldSlots.pants],
    ["장갑", "반지2", oldSlots.gloves],
  ] as const;

  const newIlvls: number[] = [];
  for (const [leftName, rightName, src] of armorPairs) {
    const mapped = mapNormalGear(src.ilvl); // 오류/미확인 처리 생략
    // leftName: mapped + honing = src.honing
    // rightName: mapped (오른쪽은 슬롯 재련 없음)
    newIlvls.push(mapped.itemLevel, mapped.itemLevel);
  }

  // 2) 무기 -> 6부위 (무기, 완갑, 허리띠, 성물, 인장, 팔찌)
  const useEstherAsBase =
    oldSlots.estherWeapon != null &&
    (!oldSlots.normalWeapon || oldSlots.estherWeapon.ilvl >= oldSlots.normalWeapon.ilvl);

  let weaponNewIlvl: number;
  let linked5Ilvl: number;
  let weaponSlotHoning: number;

  if (useEstherAsBase && oldSlots.estherWeapon) {
    const est = ESTHER_TABLE[`${oldSlots.estherWeapon.evolution}_${oldSlots.estherWeapon.ella}`];
    weaponNewIlvl = est.newIlvl;
    weaponSlotHoning = est.newHoning;
    if (oldSlots.estherWeapon.ilvl >= 1800) {
      // [미확인 주의]: "재련 25단계 무기" 기준 정확한 아이템 레벨(1800 추정 -> 1815)
      linked5Ilvl = mapNormalGear(1800).itemLevel; // 1815 (UNCONFIRMED: 상재 포함 여부 확인 필요)
    } else {
      linked5Ilvl = mapNormalGear(oldSlots.estherWeapon.ilvl).itemLevel;
    }
  } else if (oldSlots.normalWeapon) {
    const mappedW = mapNormalGear(oldSlots.normalWeapon.ilvl);
    weaponNewIlvl = mappedW.itemLevel;
    linked5Ilvl = mappedW.itemLevel;
    weaponSlotHoning = oldSlots.normalWeapon.honing;
  }

  // 무기(1) + 완갑/허리띠/성물/인장/팔찌(5)
  // 완갑 슬롯 재련 = oldSlots.oldVambraceHoning (미확인 시 사용자 입력)
  // 허리띠 슬롯 재련 = 0 (신규 재련)
  newIlvls.push(weaponNewIlvl, linked5Ilvl, linked5Ilvl, linked5Ilvl, linked5Ilvl, linked5Ilvl);

  const newAvgIlvl = newIlvls.reduce((a, b) => a + b, 0) / 16;
  return { newAvgIlvl, weaponSlotHoning, beltSlotHoning: 0 };
}

// 4. 장신구 -> 마력 주입 주문서 등급 판별 (공식 Table 5, 6)

type OptRank = "상" | "중" | "하" | "무효";

function evaluateAccessoryScroll(acc: {
  tier: number;
  grade: "고대" | "유물" | string;
  part: "목걸이" | "귀걸이" | "반지";
  quality: number;
  tradeRemainCount?: number; // 미확인 시 사용자 입력
  options: Array<{ name: string; rank: OptRank }>; // 수치 정적표로 상/중/하 판별
}) {
  if (acc.tier <= 3) return { result: "REMOVE" };
  if (acc.quality < 67) return { result: "GEM_SHARD_SILVER", note: "수량 미확인" };

  // 딜러 기준과 서포터 기준 각각 평가 가능 (장신구 자체는 거래 가능하므로 클래스 무관하게
  // 딜러/서포터 유효 옵션 조합에 따라 주문서 등급 결정; 단 두 역할 모두 만족 시 우선순위 미확인)
  function scoreAs(role: "딜러" | "서포터"): number | null {
    // 1,2옵션에 해당하는 유효 옵션 등급 추출 (opts12), 3옵션 추출 (opt3)
    const { opts12, opt3, opt3Name } = matchValidAccOptions(role, acc.part, acc.options);

    if (acc.grade === "유물") {
      if (role === "서포터" && (acc.part === "목걸이" || acc.part === "반지")
          && opts12.filter(r => r === "상").length === 2) {
        return 5; // 5등급 주문서
      }
      return null;
    }
    if (acc.grade !== "고대") return null;
    if (opts12.length === 0) return null; // 유효 옵션 없음 (단, 1줄만 있을 때 중중 이하 포함 여부는 미확인)

    const sorted12 = sortRanksDesc(opts12); // ["상","상"], ["상","중"], ...
    const isSupEarring = (role === "서포터" && acc.part === "귀걸이");

    if (isSupEarring) {
      if (sorted12[0] === "상" && sorted12[1] === "상") {
        if (opt3 === "상" && opt3Name === "최대 생명력 +") return 2;
        if (opt3 === "상" && (opt3Name === "파티원 보호막 효과 %" || opt3Name === "파티원 회복 효과 %")) return 3;
        if (opt3 === "중") return 4;
        return 5;
      }
      return 6; // 상/중 이하
    } else {
      if (sorted12[0] === "상" && sorted12[1] === "상") {
        if (opt3 === "상") return 1;
        if (opt3 === "중") return 2;
        return 3;
      }
      if (sorted12[0] === "상" && sorted12[1] === "중") return 4;
      if (sorted12[0] === "상" && sorted12[1] === "하") return 5;
      return 6; // 중중 이하
    }
  }

  const bestRank = minNonNull(scoreAs("딜러"), scoreAs("서포터"));
  if (bestRank === null) return { result: "GEM_SHARD_SILVER", note: "수량 미확인" };
  return {
    result: "MAGIC_SCROLL",
    part: acc.part,
    scrollGrade: bestRank, // 1~6등급
    tradeRemainCount: acc.tradeRemainCount ?? "UNCONFIRMED",
  };
}

function evaluateAbilityStones(
  stones: Array<{

  function isValidPair(a: string, b: string): boolean {
    const dealerOk = DEALER_VALID_ENGRAVINGS.has(a) && DEALER_VALID_ENGRAVINGS.has(b);
    const supportOk = canSupport && SUPPORT_VALID_ENGRAVINGS.has(a) && SUPPORT_VALID_ENGRAVINGS.has(b);
    return dealerOk || supportOk;
  }

  function mapToUpgrade(p1: number, p2: number) {
    const [hi, lo] = [Math.max(p1, p2), Math.min(p1, p2)];
    const sum = hi + lo;
    if (sum < 4) return null;
    if (sum === 4 && hi === 2 && lo === 2) return { slots: 1, level: 3, extra: false, score: 1 };
    if (sum === 4 && hi === 3 && lo === 1) return { slots: 1, level: 5, extra: false, score: 2 };
    if (sum === 5)                         return { slots: 2, level: 7, extra: false, score: 3 };
    if (sum === 6 && hi === 4 && lo === 2) return { slots: 3, level: 7, extra: false, score: 4 };
    if (sum === 6 && hi === 3 && lo === 3) return { slots: 4, level: 7, extra: false, score: 5 };
    if (sum === 7)                         return { slots: 5, level: 7, extra: false, score: 6 };
    if (sum >= 8)                          return { slots: 5, level: 7, extra: true,  score: 7 };
    return null;
  }
